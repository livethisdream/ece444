# Where is the HB100 under Calibrate's settings? Run on the Pi, kit idle.
import os, sys, time, numpy as np
os.chdir("/home/analog/pyadi-iio/examples/phaser")
sys.path.insert(0, os.getcwd())
import config
from phaser_functions import spec_est, load_hb100_cal
from adi import ad9361
from adi.cn0566 import CN0566

F = load_hb100_cal()
print(f"stored HB100 {F/1e9:.6f} GHz   config Rx_freq {config.Rx_freq/1e9:.3f} GHz")
p = CN0566(uri="ip:localhost"); s = ad9361(uri="ip:192.168.2.1"); p.sdr = s
time.sleep(0.5); p.configure(device_mode="rx")
p.SDR_init(30000000, config.Tx_freq, config.Rx_freq, 6, -6, 1024)
s.tx_hardwaregain_chan0 = s.tx_hardwaregain_chan1 = -88
p.freq_dev_range = 0; p.freq_dev_time = 0; p.powerdown = 0; p.ramp_mode = "disabled"
for i in range(8): p.set_chan_gain(i, 127, apply_cal=False); p.set_chan_phase(i, 0, apply_cal=False)
fs = int(s.sample_rate)
print(f"fs {fs}")
print("  Rx_freq_GHz   LO_GHz   peak_dBFS  peak_offset_MHz   all_off_dBFS")
for rx in (1.9e9, 2.0e9, 2.1e9, 2.2e9, 2.3e9):
    s.rx_lo = int(rx)
    p.frequency = (int(F) + int(rx)) // 4
    def grab():
        time.sleep(0.05); s.rx_destroy_buffer(); d = s.rx()
        a, fr = spec_est(d[0] + d[1], fs, ref=2 ** 11, plot=False)
        a, fr = np.fft.fftshift(a), np.fft.fftshift(fr)
        k = int(np.argmax(a)); return a[k], fr[k]
    on, off_hz = grab()
    for i in range(8): p.set_chan_gain(i, 0, apply_cal=False)
    off, _ = grab()
    for i in range(8): p.set_chan_gain(i, 127, apply_cal=False)
    print(f"    {rx/1e9:.2f}       {(F+rx)/1e9:.4f}    {on:7.1f}      {off_hz/1e6:+7.3f}        {off:7.1f}")
