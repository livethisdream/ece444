% Module banner: the one heavy element on the page, so a module is findable
% at arm's length. Sub-heads are light by comparison -- they group, not divide.
\newcommand{\modhead}[1]{%
  \vspace{1pt}%
  {\normalfont\sffamily\normalsize\bfseries #1}\\[1.5pt]%
  \hrule height 0.9pt%
  \vspace{5pt}%
}
% Sub-heads are bold sans with a trailing rule to the column edge. The rule is
% light gray -- black!45, the same value as the column separators -- against the
% module banner's solid 0.9pt black, so the three levels separate by tone as
% well as weight: banner, sub-head, body.
%
% \hrulefill inside a color group is what does it. A measured-width \rule also
% takes the color, but sizing it to the column made every heading fragile

\modhead{Module 1 \quad Foundations}

\grp{Waves and power}
\def\linespacing{5pt}
\eqL{
&k = \dfrac{2\pi}{\lambda} &\\[\linespacing]
&\eta_0 = \sqrt{\dfrac{\mu_0}{\epsilon_0}} \approx 377\ \ohms &\\[\linespacing]
&S = \dfrac{\vert \mathbf{E}\vert^2}{2\eta_0} &\\[\linespacing]
&U(\theta,\phi) = r^2 S &\\[\linespacing]
&P_\text{rad} = \oint U(\theta,\phi)\, d\Omega &
}

\grp{Directivity and gain}
\def\linespacing{5pt}
\eqL{
&D = \dfrac{4\pi U_\text{max}}{P_\text{rad}} &\\[\linespacing]
&D \approx \dfrac{41{,}253}{\theta_E^\circ\, \theta_H^\circ} &\\[\linespacing]
&\Omega_A = \dfrac{P_\text{rad}}{U_\text{max}} &\\[\linespacing]
&D = \dfrac{4\pi}{\Omega_A} &\\[\linespacing]
&\eta_\text{rad} = \dfrac{R_\text{rad}}{R_\text{rad} + R_\text{loss}} &\\[\linespacing]
&G = \eta_\text{rad}\, D &\\[\linespacing]
&A_e = \dfrac{\lambda^2 G}{4\pi} &
}

\grp{Polarization}
\def\linespacing{5pt}
\eqL{
&\text{AR} = \dfrac{E_\text{major}}{E_\text{minor}} &\\[\linespacing]
&\text{PLF} = \vert \myhat{\rho}_w \mycdot \myhat{\rho}_a \vert^2 &\\[\linespacing]
&\text{PLF} = \cos^2\psi \quad \text{(two linear, tilt }\psi) &
}

\grp{Impedance and matching}
\def\linespacing{4pt}
\eqL{
&Z_\text{in} = R_\text{rad} + R_\text{loss} + jX_\text{in} &\\[\linespacing]

&\text{ML} = -10\log_{10}\lp 1 - \vert\Gamma\vert^2 \rp &\\[\linespacing]
&Z_1 = \sqrt{Z_0 Z_L} &\\[\linespacing]
&Q = \sqrt{\dfrac{R_\text{high}}{R_\text{low}} - 1} &
}

\grp{Field regions}
\def\linespacing{4pt}
\eqL{
&r = \dfrac{\lambda}{2\pi} &\\[\linespacing]
&r < 0.62\sqrt{\dfrac{D^3}{\lambda}} &\\[\linespacing]
&r \geq \dfrac{2D^2}{\lambda} &
}

\grp{Radiation integral}
\def\linespacing{4pt}
\eqL{
&\mathbf{A} = \dfrac{\mu}{4\pi}\int \mathbf{J}\, \dfrac{e^{-jkR}}{R}\, dV^{\prime} &\\[\linespacing]
&R \approx r - \myhat{r}\mycdot\mathbf{r}^{\prime} &\\[\linespacing]
&\mathbf{N} = \int \mathbf{J}\, e^{jk\myhat{r}\mycdot\mathbf{r}^{\prime}}\, dV^{\prime} &\\[\linespacing]
&E_\theta = -j\omega\mu\, \dfrac{e^{-jkr}}{4\pi r}\, N_\theta &
}

\grp{Link budget}
\def\linespacing{4pt}
\eqL{
&\text{EIRP} = P_t G_t &\\[\linespacing]
&\text{FSPL} = 20\log_{10}\lp \dfrac{4\pi R}{\lambda} \rp &\\[\linespacing]
&P_r = P_t G_t G_r \lp \dfrac{\lambda}{4\pi R} \rp^2 &
}

\modhead{Module 3 \quad Arrays and Beamforming}

\grp{Array factor}
\def\linespacing{8pt}
\eqL{
&\text{AF}(\psi) = \sum_{n=0}^{N-1} a_n e^{jn\psi} &\\[\linespacing]
&\psi = kd\sin\theta + \beta &\\[\linespacing]
&\vert \text{AF} \vert = \dfrac{\sin(N\psi/2)}{N\sin(\psi/2)} &\\[\linespacing]
&F(\theta) = \text{EF}(\theta) \cdot \text{AF}(\theta) &
}

\grp{Beam steering}
\def\linespacing{8pt}
\eqL{
&\beta = -kd\sin\theta_0 &\\[\linespacing]
&\Delta\phi = 360^\circ \lp \dfrac{d}{\lambda} \rp \sin\theta_0 &\\[\linespacing]
&\beta = 0 \ \text{(broadside)} \qquad \beta = -kd \ \text{(endfire)} &
}

\grp{Beamwidth and directivity}
\def\linespacing{8pt}
\eqL{
&\sin\theta_n = \dfrac{n\lambda}{Nd} &\\[\linespacing]
&\theta_\text{HP} = \dfrac{0.886\,\lambda}{Nd} &\\[\linespacing]
&\theta_\text{HP}(\theta_0) = \dfrac{\theta_\text{HP}(0)}{\cos\theta_0} &\\[\linespacing]
&D = \dfrac{2Nd}{\lambda} &
}

\grp{Gain and scan loss}
\def\linespacing{8pt}
\eqL{
&G(\theta_0) = G_e(\theta_0) + 10\log_{10} N &\\[\linespacing]
&G_e(\theta_0) = G_e(0) + 10\log_{10}\lp \cos\theta_0 \rp &
}
