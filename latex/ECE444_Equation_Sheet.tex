\textbf{Wave Fundamentals and the Far Field}
\def\linespacing{3pt}
\eq{
&k = \dfrac{2\pi}{\lambda}, \quad c = 3\ee{8}\ \text{m/s} \\[\linespacing]
&\eta_0 = \dfrac{\vert\mathbf{E}\vert}{\vert\mathbf{H}\vert} \approx 377\ \ohms\text{ (free space)} \\[\linespacing]
&\text{outgoing wave: } \dfrac{e^{-jkr}}{r};\ \text{field} \propto \dfrac{1}{r},\ \text{power} \propto \dfrac{1}{r^2} \\[\linespacing]
&P_\text{avg} = \dfrac{1}{2}\re{\mathbf{E}\times\mathbf{H}^*}; \quad S_\text{rad} = \dfrac{\vert\mathbf{E}\vert^2}{2\eta_0} \\[\linespacing]
&U(\theta,\phi) = r^2 S_\text{rad}\text{ (W/sr); whole sphere} = 4\pi\ \text{sr}
}

\vspace{4pt}
\textbf{Pattern, Directivity, Gain}
\def\linespacing{3pt}
\eq{
&\vert F(\theta,\phi)\vert\text{ normalized field pattern; isotropic: } D{=}1,\ 0\ \dbi \\[\linespacing]
&D(\theta,\phi) = \dfrac{4\pi U}{P_\text{rad}};\ \text{pencil beam} \approx \dfrac{41{,}253}{\theta_1^\circ\theta_2^\circ}\ \text{(real 26k--32k)} \\[\linespacing]
&G = \eta_\text{rad} D,\ \eta_\text{rad} = \dfrac{P_\text{rad}}{P_\text{in}}; \quad \text{dBi} = \text{dBd} + 2.15 \\[\linespacing]
&G_\text{re} = \lp 1-\vert\Gamma\vert^2\rp G;\ \text{VSWR 2:1 costs } 11\%\ (\approx 0.5\ \db) \\[\linespacing]
&\theta_\text{HP}\text{ (HPBW) at} -3\ \db;\quad \text{FNBW} \approx 2\ \theta_\text{HP} \\[\linespacing]
&A_e = \dfrac{\lambda^2 G}{4\pi} = \eta_\text{ap} A_\text{phys} \\[\linespacing]
&\text{reciprocity: } A_e/G = \lambda^2/4\pi\text{ for every antenna}
}

\def\linespacing{3pt}

&\text{wrong circular sense: AR}{=}3\ \db{\to}{-}9.6\ \db,\ \text{AR}{=}6\ \db{\to}{-}4.5\ \db \\[\linespacing]

}

\textbf{Impedance, VSWR, Matching}
\def\linespacing{3pt}
\eq{
&Z_\text{in} = R_\text{in} + jX_\text{in}\text{ (resonance: } X_\text{in}{=}0\text{)}; \quad \eta_\text{rad} = \dfrac{R_\text{rad}}{R_\text{rad}+R_\text{loss}} \\[\linespacing]
&\Gamma = \dfrac{Z_\text{in}-Z_0}{Z_\text{in}+Z_0}; \quad \text{VSWR} = \dfrac{1+\vert\Gamma\vert}{1-\vert\Gamma\vert} \\[\linespacing]
&\text{VSWR}{\le}2 \leftrightarrow \text{RL}{\ge}9.5\ \db \leftrightarrow 11\%\text{ reflected} \\[\linespacing]
&\text{mismatch loss} = -10\log_{10}\lp1-\vert\Gamma\vert^2\rp \le 0.5\ \db\text{ at VSWR}{=}2 \\[\linespacing]
&Z_\text{in} = Z_0\dfrac{Z_L + Z_0\tanh\lp\gamma\ell\rp}{Z_0 + Z_L\tanh\lp\gamma\ell\rp};\text{ lossless: tan for tanh} \\[\linespacing]
&\Gamma(z) = \Gamma_L e^{2\gamma z}; \quad \beta = \omega\sqrt{\mu\epsilon} = 2\pi/\lambda \\[\linespacing]
&\text{quarter-wave transf.: } Z_1 = \sqrt{Z_0 R_L} \\[\linespacing]
&\text{L-match } Q = \sqrt{R_\text{big}/R_\text{small} - 1};\ \text{balun 4:1: } 300\ \ohms \to 75\ \ohms
}

\textbf{Field Regions}
\def\linespacing{3pt}
\eq{

&\text{true far field: } kr \gg 1 \\[\linespacing]
&\text{reactive near-field edge: } r = 0.62\sqrt{D^3/\lambda} \\[\linespacing]
&\text{far-field distance: } r \ge \dfrac{2D^2}{\lambda}\ (D{=}\text{largest dim}) \\[\linespacing]
&\text{phase-error budget: } \Delta \le \lambda/16,\ \pi/8\ (22.5^\circ)
}

\textbf{Radiation Integrals}

}
