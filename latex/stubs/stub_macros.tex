\usepackage[margin=1in]{geometry}
\usepackage{amsmath,amssymb}
\usepackage{graphicx}
\usepackage{xcolor}
\usepackage{circuitikz}

% ---- units and short forms (spacing matches the reference PDFs) ----
\newcommand{\ohms}{\ensuremath{\,\Omega}}
\newcommand{\m}{\ensuremath{\,\mathrm{m}}}
\newcommand{\ghz}{\ensuremath{\,\mathrm{GHz}}}
\newcommand{\mhz}{\ensuremath{\,\mathrm{MHz}}}
\newcommand{\db}{\ensuremath{\,\mathrm{dB}}}
\newcommand{\dbi}{\ensuremath{\,\mathrm{dBi}}}
\newcommand{\mydeg}{\ensuremath{^{\circ}}}

% ---- operators written with parentheses ----
\newcommand{\cosp}[1]{\ensuremath{\cos\!\left(#1\right)}}
\newcommand{\sinp}[1]{\ensuremath{\sin\!\left(#1\right)}}
\newcommand{\mylog}[1]{\ensuremath{\log_{10}\!\left(#1\right)}}

% ---- vectors and delimiters ----
\newcommand{\xhat}{\ensuremath{\hat{\mathbf{x}}}}
\newcommand{\yhat}{\ensuremath{\hat{\mathbf{y}}}}
\newcommand{\zhat}{\ensuremath{\hat{\mathbf{z}}}}
\newcommand{\lp}{\left(}
\newcommand{\rp}{\right)}

% ---- display equation; \eq{a &= b \\ &= c} has to accept alignment ----
\newcommand{\eq}[1]{\begin{align*}#1\end{align*}}

% \fullwidth is exam.cls's own -- do not redefine it here.

% ---- boxed final answer: empty on the blank copy, filled in red on the key ----
\newcommand{\ansbox}[2][1.3in]{%
  \fbox{\parbox[c][0.32in][c]{#1}{\centering\ifprintanswers\color{red}#2\else\phantom{#2}\fi}}}

% ---- solution styling, matched to the committed L02/L03/L04 PDFs ----
% The reference sets show worked solutions as plain red text: no frame, no
% "Solution:" heading. exam.cls frames and titles them by default, which would
% inflate every solution's height and make the page-break check meaningless.
\unframedsolutions
\renewcommand{\solutiontitle}{}

\newcommand{\watts}{\ensuremath{\,\mathrm{W}}}
\newcommand{\dbm}{\ensuremath{\,\mathrm{dBm}}}
