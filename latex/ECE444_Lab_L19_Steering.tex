\question \textbf{The prediction table.} Every phase in this lab depends on the
source frequency. Use the HB100 frequency your kit reported in Lesson 17. If you
did not record one, use the nominal $10.525\ \ghz$ and say so.

\begin{parts}

	\begin{minipage}[t]{0.58\linewidth}
		\part Record the frequency you are using, then find the wavelength, the
		electrical spacing $d/\lambda$, and $kd$ in degrees.
		\begin{solution}[1.5in]
		The key uses the nominal $10.525\ \ghz$:
		\eq{ \lambda &= \frac{c}{f} = \frac{2.998\times 10^{8}}{10.525\times 10^{9}} = 0.02848\ \m = 28.48\ \text{mm} \\
		     \frac{d}{\lambda} &= \frac{14}{28.48} = 0.4915 \\
		     kd &= 360\mydeg\ \frac{d}{\lambda} = 176.9\mydeg }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$f$ = \ansbox[1.3in]{$10.525\ \ghz$}\\[10pt]
		$d/\lambda$ = \ansbox[1.3in]{$0.4915$}\\[10pt]
		$kd$ = \ansbox[1.3in]{$176.9\mydeg$}
	\end{minipage}

	\part Find the progressive phase $\Delta\phi$ that steers the beam to each
	angle in the table.

	\begin{center}
	\renewcommand{\arraystretch}{1.25}
	\begin{tabular}{| l | K{0.9in} | K{0.9in} | K{0.9in} | K{0.9in} |}
	\hline
	$\theta_0$ & $0\mydeg$ & $15\mydeg$ & $30\mydeg$ & $45\mydeg$ \\\hline
	$\Delta\phi$ & \kv{$0.0\mydeg$} & \kv{$45.8\mydeg$} & \kv{$88.5\mydeg$} & \kv{$125.1\mydeg$} \\\hline
	\end{tabular}
	\end{center}

		\begin{solution}[0.6in]
		$\Delta\phi = kd\sinp{\theta_0} = 176.9\mydeg\sinp{\theta_0}$; at $30\mydeg$,
		$176.9\mydeg \times 0.500 = 88.5\mydeg$.
		\end{solution}

	\newpage
	\part Fill in the $30\mydeg$ ramp. Lesson 18 numbers the elements $n = 0$ to
	7, so element E$(n+1)$ is set to $n\Delta\phi$. Give each value unwrapped, then
	wrapped into $0\mydeg$ to $360\mydeg$, and name the first element that wraps.

	\begin{center}
	\small
	\setlength{\tabcolsep}{4pt}
	\renewcommand{\arraystretch}{1.25}
	\begin{tabular}{| l | K{0.48in} | K{0.48in} | K{0.48in} | K{0.48in} | K{0.48in} | K{0.48in} | K{0.48in} | K{0.48in} |}
	\hline
	\textbf{Element} & \textbf{E1} & \textbf{E2} & \textbf{E3} & \textbf{E4} & \textbf{E5} & \textbf{E6} & \textbf{E7} & \textbf{E8} \\\hline\hline
	Unwrapped & \kv{$0.0$} & \kv{$88.5$} & \kv{$176.9$} & \kv{$265.4$} & \kv{$353.9$} & \kv{$442.4$} & \kv{$530.8$} & \kv{$619.3$} \\\hline
	Wrapped & \kv{$0.0$} & \kv{$88.5$} & \kv{$176.9$} & \kv{$265.4$} & \kv{$353.9$} & \kv{$82.4$} & \kv{$170.8$} & \kv{$259.3$} \\\hline
	\end{tabular}
	\end{center}

		\begin{solution}[0.6in]
		E6 is the first to wrap: $5(88.47\mydeg) = 442.4\mydeg$, and
		$442.4\mydeg - 360\mydeg = 82.4\mydeg$. Each entry comes from the unrounded
		$88.47\mydeg$, so E8 is $7(88.47\mydeg) = 619.3\mydeg$.
		\end{solution}

	\part E6 carries $442.4\mydeg$ in one table and $82.4\mydeg$ in the other. In
	two or three sentences, explain why both settings give the same pattern, and
	state what the wrapped table no longer records.
		\begin{solution}[1.2in]
		A phase of $360\mydeg$ is one full cycle of the carrier, so the element's
		weight $e^{-j\phi}$ is the same complex number for $442.4\mydeg$ and
		$82.4\mydeg$, and every term of the array sum is unchanged. What the
		wrapped table loses is the true delay: the unwrapped ramp is a delay in
		seconds that is correct at every frequency, while the wrapped values
		reproduce it only at the frequency used for $k$. That is why a wideband
		array uses time-delay units instead of phase shifters alone.
		\end{solution}

\end{parts}

	\part During a beam sweep the backend adds its own ramp, $+n\Delta\phi_s$, to
	whatever Phase Control holds, and the plot's x-axis is the angle that
	$\Delta\phi_s$ alone would steer to. With the source at boresight and the
	Apply $30$ offsets loaded, at which plotted angle does the total ramp vanish?
	Use your answer to explain, in one or two sentences, why Apply writes the
	ramp with a minus sign.
		\begin{solution}[1.5in]
		Element $n$ carries $n\Delta\phi_s - n\Delta\phi_0 = n(\Delta\phi_s - \Delta\phi_0)$.
		A boresight source needs no ramp, so the array points at it when
		$\Delta\phi_s = \Delta\phi_0$, which is the plotted angle $+30\mydeg$. The
		minus sign is what makes a boresight source's trace peak at $+\theta_0$ on
		the sweep plot. On their own, with no sweep ramp added, the offsets
		$-n\Delta\phi_0$ would point the beam to $-30\mydeg$.
		\end{solution}

\question \textbf{The beam and the error budget.} Continue with your frequency
from Question 1. The source sits on a protractor arc $1.00\ \m$ from the array.

\begin{parts}

	\part Use $\theta_{\text{HP}} \approx 0.886\lambda/(Nd\cos\theta_0)$ to predict
	the half-power beamwidth, and $10\mylog{\cos\theta_0}$ to predict the drop in
	peak level relative to broadside (the ideal-element scan loss).

	\begin{center}
	\renewcommand{\arraystretch}{1.25}
	\begin{tabular}{| l | K{1.1in} | K{1.1in} | K{1.1in} |}
	\hline
	$\theta_0$ & $0\mydeg$ & $30\mydeg$ & $45\mydeg$ \\\hline
	$\theta_{\text{HP}}$ & \kv{$12.9\mydeg$} & \kv{$14.9\mydeg$} & \kv{$18.3\mydeg$} \\\hline
	Scan loss & \kv{$0$ dB} & \kv{$-0.6$ dB} & \kv{$-1.5$ dB} \\\hline
	\end{tabular}
	\end{center}

		\begin{solution}[0.9in]
		$0.886(28.48)/112 = 0.2253\ \text{rad} = 12.9\mydeg$ at broadside, divided
		by $\cosp{\theta_0}$ off broadside. The exact array factor gives
		$13.0\mydeg$, $15.1\mydeg$, and $18.7\mydeg$; the rule reads a few tenths
		narrow. A real patch element rolls off faster than the ideal element, so
		expect $-0.8$ to $-0.9$ dB at $30\mydeg$ on the kit.
		\end{solution}

	\begin{minipage}[t]{0.58\linewidth}
		\part Lab Preset 1 sweeps by stepping $\Delta\phi_s$ one phase-shifter LSB,
		$2.8125\mydeg$, at a time. One step moves $\sin\theta$ by
		$2.8125\mydeg/kd$. Find the angle step at $0\mydeg$ and at $30\mydeg$, and the
		worst-case error of a peak read off this grid at $30\mydeg$.
		\begin{solution}[1.5in]
		\eq{ \Delta(\sin\theta) &= \frac{2.8125\mydeg}{176.9\mydeg} = 0.0159 \\
		     \Delta\theta &= \frac{0.0159}{\cos\theta}\ \text{rad} = 0.91\mydeg \text{ at } 0\mydeg,\ 1.05\mydeg \text{ at } 30\mydeg }
		The trace is sampled only at grid points, so the peak can be off by half a
		step: $0.53\mydeg$ at $30\mydeg$.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		step at $0\mydeg$ = \ansbox[1.1in]{$0.91\mydeg$}\\[10pt]
		step at $30\mydeg$ = \ansbox[1.1in]{$1.05\mydeg$}\\[10pt]
		half step = \ansbox[1.1in]{$0.53\mydeg$}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part How far along the arc is one degree? If you can place the tripod to
		within $25\ \text{mm}$, how large is the error in the reference angle?
		\begin{solution}[1.1in]
		\eq{ s &= R\ \theta_{\text{rad}} = (1000\ \text{mm})\frac{\pi}{180} = 17.5\ \text{mm per degree} \\
		     \frac{25}{17.5} &= 1.4\mydeg }
		The budget uses $1.5\mydeg$, which also covers centering the arc and pointing the module.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		per degree = \ansbox[1.1in]{$17.5\ \text{mm}$}\\[10pt]
		aim error = \ansbox[1.1in]{$1.4\mydeg$}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part A reflection from the bench arrives $20\ \db$ below the direct path,
		an amplitude ratio of $0.1$. Find the largest and smallest total level it
		can produce, in dB relative to the direct path alone.
		\begin{solution}[1.0in]
		\eq{ 20\mylog{1 + 0.1} &= +0.8\ \db \\ 20\mylog{1 - 0.1} &= -0.9\ \db }
		Depending on its phase, the reflection puts that ripple on the main lobe,
		which moves the apparent peak by up to about $1\mydeg$.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		largest = \ansbox[1.1in]{$+0.8\ \db$}\\[10pt]
		smallest = \ansbox[1.1in]{$-0.9\ \db$}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part Combine the three independent peak-angle errors at $30\mydeg$ (grid
		half step, aim $1.5\mydeg$, multipath $1.0\mydeg$) as a root sum of
		squares, $\sqrt{a^2 + b^2 + c^2}$. State a pass standard for the lab in
		degrees, and name the term you could reduce most.
		\begin{solution}[1.3in]
		\eq{ \sqrt{0.53^2 + 1.5^2 + 1.0^2} = 1.9\mydeg }
		A measured peak within $\pm 2\mydeg$ of the reference passes. The aim term
		is the largest, and careful placement of the tripod reduces it; the grid
		term is fixed by the 7-bit phase shifter.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		RSS = \ansbox[1.1in]{$1.9\mydeg$}\\[10pt]
		pass = \ansbox[1.1in]{$\pm 2\mydeg$}
	\end{minipage}

\end{parts}

\question \textbf{Reconcile the measurements.} Answer each part in three or four
sentences, using your team's numbers from Part 2 and your own predictions from
Part 1.

\begin{parts}

	\part Compare each Table A difference with the pass standard from
	Question 3(e). For any row outside it, name the error source in your budget
	most likely responsible, and say how you would check it.
	\begin{solutionorlines}[1.5in]
	Every row should sit within $\pm 2\mydeg$. A row outside it most often comes
	from aim: the tripod off its mark, the arc not centered on the middle of the
	array (a $10\ \text{mm}$ offset moves the reference about $0.5\mydeg$), or the
	module not pointed back at the array. A shift common to all three rows points
	at the $0\mydeg$ placement or at calibration; a single row that moves when
	people step away from the arc points at multipath. The grid alone can
	contribute only half a step, $0.46$ to $0.65\mydeg$.
	\end{solutionorlines}

	\part Compare the $30\mydeg$ rows of Tables A and B: Est. Angle, HPBW, and Peak
	Array Gain. Explain why the beamwidths agree but only the physical run loses
	peak level.
	\begin{solutionorlines}[1.8in]
	Both traces have the same shape because the array factor depends only on
	$\sin\theta - \sin\theta_0$ (Lesson 18): moving the source to $30\mydeg$ and
	offsetting the ramp by $30\mydeg$ shift the same function to the same place,
	so both widen to about $15\mydeg$. The sweep holds the element pattern fixed
	at the source's direction, so the trace is the array factor scaled by
	$EF(\theta_{\text{src}})$. In the physical run the source sits $30\mydeg$ off
	the elements' boresight and every point drops by the element roll-off,
	about $0.6$ to $0.9$ dB. In the commanded run the source stays at boresight,
	$EF$ stays at its peak, and the peak level matches the $0\mydeg$ row.
	\end{solutionorlines}

	\part Compare the Phase Control read-back with your prediction. Explain the
	sign in your own words, and account for any nonzero difference.
	\begin{solutionorlines}[1.5in]
	Apply writes $-n\Delta\phi_0$ so that the sweep's own $+n\Delta\phi_s$ ramp
	cancels it at the plotted angle $\theta_0$, which is where a boresight source
	then peaks; the offsets alone would point the beam to $-\theta_0$. The GUI
	rounds to whole degrees, so a prediction made at the same frequency and
	rounded the same way matches exactly, and an unrounded one differs by at most
	$0.5\mydeg$. A difference that
	grows with element number means the prediction used a different frequency
	from the Signal Freq on screen.
	\end{solutionorlines}

	\part Compare each measured HPBW with its calculated value, and explain
	why the lab quotes beamwidth to the nearest degree.
	\begin{solutionorlines}[1.3in]
	The measured widths should sit within about a degree of $13.0\mydeg$,
	$15.1\mydeg$, and $18.7\mydeg$. The trace is sampled every $0.9$ to
	$1.3\mydeg$, so each half-power crossing falls between two samples and is
	read off a straight line drawn between them. That limits each crossing to a
	fraction of a step, and the width to about a degree.
	\end{solutionorlines}

\end{parts}

\question \textbf{Reading a measured sweep.} Another team places the HB100 at
$+30.0\mydeg$ on the arc with Phase Control at zero. The trace peaks at an Est.
Angle of $28.5\mydeg$, its half-power crossings are at $20.5\mydeg$ and
$35.8\mydeg$, its peak sits $0.7\ \db$ below the boresight trace, and its first
sidelobes read $-11.8$ dB relative to its own peak.
\begin{parts}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the measured half-power beamwidth and compare it with the
		$1/\cos\theta_0$ prediction from the $13.0\mydeg$ broadside value.
		\begin{solution}[1.1in]
		\eq{ \theta_{\text{HP,meas}} &= 35.8\mydeg - 20.5\mydeg = 15.3\mydeg \\
		     \frac{13.0\mydeg}{\cosp{30\mydeg}} &= \frac{13.0\mydeg}{0.866} = 15.0\mydeg }
		The measured width is $0.3\mydeg$ wider than predicted, well inside one
		grid step ($1.05\mydeg$ near $30\mydeg$).
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$\theta_\text{HP}$ = \ansbox[1.3in]{$15.3\mydeg$}\\[10pt]
		predicted = \ansbox[1.3in]{$15.0\mydeg$}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the difference between Est. Angle and the arc angle, and express
		it in grid steps.
		\begin{solution}[0.9in]
		\eq{ \Delta\theta &= 28.5\mydeg - 30.0\mydeg = -1.5\mydeg \\
		     \frac{1.5\mydeg}{1.05\mydeg} &= 1.4 \text{ grid steps} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		offset = \ansbox[1.3in]{$1.4$ steps}
	\end{minipage}

	\part The team reports the array as broken because the peak did not land on
	$30\mydeg$. In two or three sentences, say whether the data support that
	conclusion.
		\begin{solution}[1.1in]
		They do not. The $1.5\mydeg$ offset is more than the grid's half step can
		produce, but it sits inside the $1.9\mydeg$ budget and the $\pm 2\mydeg$
		standard, and the aim term alone is about $1.5\mydeg$. A fault in the array
		would show as an offset in the $0\mydeg$ row as well, or as a beamwidth or
		sidelobe level far from the prediction, and these read $15.3\mydeg$ and
		$-11.8$ dB.
		\end{solution}

	\part The peak fell $0.7\ \db$, while the ideal-element bound is
	$-0.6\ \db$. Give the most likely reason the measured drop is the larger, and
	say whether you would expect the gap to grow or shrink at $50\mydeg$.
		\begin{solution}[1.2in]
		The bound assumes an element that accounts only for the foreshortened
		cell, a $\cos\theta$ power pattern. A real patch rolls off faster, closer
		to $\cos^{1.3}\theta$ to $\cos^{1.5}\theta$ in power, so the measured drop
		is at least the bound. The gap grows with angle: at $50\mydeg$ the bound
		is $-1.9\ \db$ and a $\cos^{1.3}$ to $\cos^{1.5}$ element gives $-2.5$ to
		$-2.9\ \db$, a gap of $0.6$ to $1.0\ \db$ against $0.2$ to $0.3\ \db$ at
		$30\mydeg$.
		\end{solution}

\end{parts}

The boresight trace is $13.1\mydeg$ wide at half power and sets the $0\ \db$

wide at half power, and its peak is $1.7\ \db$ below the boresight peak.
\begin{parts}

	\begin{minipage}[t]{0.58\linewidth}
		\part Predict the half-power width at $45\mydeg$ from the measured
		boresight width, and compare it with the measurement.
		\begin{solution}[1.2in]
		\eq{ \theta_\text{HP} \approx \frac{13.1\mydeg}{\cosp{45\mydeg}} = \frac{13.1\mydeg}{0.707} = 18.5\mydeg }
		The trace reads $0.4\mydeg$ wider, inside one grid step ($1.3\mydeg$ near
		$45\mydeg$). From the theory width, $12.9\mydeg$, the rule gives
		$18.3\mydeg$, and the exact array factor gives $18.7\mydeg$.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$\theta_\text{HP}$ = \ansbox[1.3in]{$18.5\mydeg$}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part Predict the peak drop from the ideal-element bound and compare it
		with the measured $1.7\ \db$.
		\begin{solution}[1.0in]
		\eq{ \Delta G &= 10\mylog{\cosp{45\mydeg}} \\ &= 10\mylog{0.707} = -1.5\ \db }
		The measured drop is $0.2\ \db$ deeper, consistent with a patch element
		that rolls off faster than the ideal one.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$\Delta G$ = \ansbox[1.3in]{$-1.5\ \db$}
	\end{minipage}

	\part Name the geometric fact behind the wider beam, and the one behind the
	lower peak.
		\begin{solution}[1.2in]
		The beam widens because a source at $45\mydeg$ sees the array's projected
		length, $Nd\cosp{\theta_0}$, which is shorter than $Nd$. The peak drops
		because each element's own cell presents a projected area that shrinks as
		$\cosp{\theta}$, so the element pattern rolls off; the array factor's
		directivity stays near 7.7 as it scans (Lesson 18, Scan Loss).
		\end{solution}

	\begin{minipage}[t]{0.58\linewidth}
		\part A specification calls for a $13\mydeg$ beam held out to $45\mydeg$ of
		scan. How many elements would the array need at the same
		$14\ \text{mm}$ spacing? Name the hardware tradeoff of that choice in one
		sentence.
		\begin{solution}[1.2in]
		The projected aperture at $45\mydeg$ must equal today's broadside aperture:
		\eq{ N\cosp{45\mydeg} &= 8 \\ N &= \frac{8}{0.707} = 11.3 \rightarrow 12 \text{ elements} }
		The tradeoff is hardware: twelve elements need twelve LNAs and phase
		shifters and three ADAR1000 chips instead of two, and the larger array
		still loses the same $1.5\ \db$ of scan loss.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$N$ = \ansbox[1.3in]{$12$ elements}
	\end{minipage}

\end{parts}
