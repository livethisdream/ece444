\question{A load \myz{L} is to be attached to a source with impedance \myz{s}. Design a transmission line network to match the load to the source with impedance \myz{s}. The matching network should consist of a length of transmission line ($\ell_{\text{tx line}}$) with characteristic impedance \myz{0,1} and an \textbf{open circuit} stub tuner (placed at $z=-\ell_{\text{tx line}}$). The stub is comprised of a length of transmission line ($\ell_{\text{stub}}$) with characteristic impedance \myz{0,stub}. }

\vspace{12pt}

\begin{parts}

\part Plot the normalized impedance, $z_{\text{L}}$ on the Smith Chart. Write that normalized impedance in the box below. 

	\hfill%
	$z_{\text{L}}$ = \ansbox[1.5in]{$1.2-j1.5 $}
	
\part Using the Smith Chart, measure the load reflection coefficient.

	\hfill% 
	$\Gamma_L $ = \ansbox[1.5in]{$0.57\angle{-47\mydeg}$}

	\part Very the load reflection coefficient by calculation. 
	
			\begin{solution}[0.25in]
				\eq{
					\Gamma_L &= \dfrac{1.2 - j1.5 -1 }{1.2-j1.5+1} = 0.568\angle{-48.1\mydeg}
				}
			\end{solution}
			\hfill%
			$\Gamma_L $ = \ansbox[1.5in]{$0.568\angle{-48.1\mydeg}$}
	
	\part Find the normalized admittance, $y_{\text{L}}$ on the Smith Chart and verify it mathematically.
	
			\begin{solution}[0.5in]
				\eq{
					y_{\text{L}} = \dfrac{1}{z_{\text{L}}} = \dfrac{1}{1.2-1.5} = 0.325+j0.407
				}
			\end{solution}
			
			\hfill%
			$y_{\text{L}}$ = \ansbox[1.5in]{$0.325+j0.407$}
	
	\newpage
	\part What should the real part of the input \textbf{admittance} be to match the transmission line to the source impedance?  
	
		\begin{solution}[0.5in]
		
		The input impedance at the input to the 100\ohms transmission line needs to match the first transmission line. Therefore Z$_{\text{in}}$ = 75\ohms. Therefore, the normalized input impedance is:
		
	\eq{
		z_{\text{in}}=\dfrac{\myz{s}}{\myz{0,1}}  = \dfrac{75\ohms}{100\ohms} = 0.75
	} 
	
	Now, the input admittance is 
	
	\eq{
		y_{\text{in}} &= \dfrac{1}{z_{\text{in}}} = \dfrac{1}{0.75} = 1.33
	}
		
		\end{solution}
		\hfill%
		Re$\left\{ y_{\text{in}} \right\}$ = \ansbox[1.5in]{$1.33$}
	
	\part Determine the shortest distance, $\ell_{\text{tx line}}$ (in wavelengths) from the load that is required for the real part of the input \textbf{admittance} to match your answer to the previous question. 
	
		\hfill%
		$\ell_{\text{tx line}}$ = \ansbox[1.5in]{$0.121\lambda$}
		
		\vspace{12pt}

	\part Mark the location of this input admittance $y_{\textbf{in}}$ on the Smith Chart. 
	
		\hfill%
		$y_{\text{in}}$ = \ansbox[1.5in]{$1.3+j1.55$}

	\part What value of \textbf{admittance} must be added in parallel at the distance, $\ell_{\text{tx line}}$, from the load, in order to match to the source impedance, \myz{s}?
		
		\vspace{12pt}
		\hfill%
		$y_{{\text{stub}}}$ = \ansbox[1.5in]{$-j1.55$}
	
	\part Determine the length, $\ell_{\text{stub}}$, of an \textbf{open circuit} transmission line tuning stub that must be added in parallel at the distance $\ell_{\text{tx line}}$ from the load, in order to match to the tx line and load to the source impedance, \myz{s}. 

		\begin{solution}[0.5in]
		$0.158\lambda$
		\end{solution}
		\hfill%
		$\ell$ = \ansbox[1.5in]{$0.341\lambda$}
		
	\newpage
	\part Verify the stub length $\ell_{\text{stub}}$ with the transmission line equation to find the input \textbf{admittance} of the stub. Remember, the stub loads the transmission line network -- this will determine the correct \myz{L} to use in the equation:
	
	\eq{
	y_{\text{in}} &= \dfrac{1}{z_{\text{in}}} = \dfrac{1 + jz_{\text{L}}\text{tan}\lp \beta \ell\rp}{z_{\text{L}} + j\text{tan}\lp \beta \ell\rp}
	}
	
		
		\begin{solution}[4in]
		
		\eq{
		y_{\text{in}} &= \dfrac{1}{z_{\text{in}}} = \dfrac{1 + j\myz{L}\text{tan}\lp \beta \ell\rp}{\myz{L} + j\text{tan}\lp \beta \ell\rp} \\
		&=\lim_{\myz{L} \to \infty}\dfrac{\dfrac{\myz{0}}{\myz{L}}+j\text{tan}\lp 2*\pi * 0.341\rp}{\dfrac{\myz{L}}{\myz{L}} + j\dfrac{\myz{0}}{\myz{L}}\text{tan}\lp2*pi*0.341\rp} = \dfrac{\text{tan}\lp 2\pi*0.341\rp}{1+j0} = -j1.554
		}
		
		\end{solution}

%\bonuspart[5] What would the stub length be if the characteristic impedance for the stub \myz{0,stub} was 50\ohms instead of 100\ohms?

	\part If, instead of a stub, you used a series component to perform the matching process, what would component would you use and what would the value be at 500MHz?
	
							
			\begin{solution}
				\eq{
					y_{\text{inductor}} &= \dfrac{1}{j\omega L} = -j1.554 \\
					\implies L &= \dfrac{1}{j\lp 2\pi \rp \lp 500\ee{6}\text{Hz} \rp \lp 100 \ohms\rp \lp -j1.554 \rp } = 2.05\text{pH}
				}
			\end{solution}
			
			\vfill%
			\hfill%
			Component = \ansbox[1.25in]{Inductor}
			
			\hfill%
			Value = \ansbox[1.25in]{2.05pH}

\fullwidth{\textbf{Learning Objective 3.7: } I can plot the transient response of a load connected to a source through a transmission line.}

\question[30]{A voltage is applied to the input of the following transmission line, which has a relative permittivity ($\varepsilon_r$) of 1.56. Accurately sketch the load voltage for at least 8$\mu$s and determine the load voltage after 3 minutes.}

\vspace{12pt}

%\begin{minipage}[c]{0.5\textwidth}
\begin{center}
\begin{circuitikz}
%	\draw (0,-2) to[vsourcesquare, a^=$12\Pi\lp\dfrac{t-0.25\text{ns}}{0.5\text{ns}}\rp$] ++(0,2)
	\draw (0,-2) to[V=20V, invert] ++(0,2)
				to[R=125\ohms] ++(2,0)
				to[tline,o-o] ++ (3,0)
				to[short] ++(0.5,0)
				to[R=75\ohms] ++(0,-2)
				to[short] ++(-0.5,0)
				to[tline,o-o] ++(-3,0)
				to[short] ++(-2,0);
	\draw(3,-1) node{\myz{0} = 50\ohms};
	\draw (2,0.5) --++ (3,0) node[midway, above] {$\ell = 480$m};

Graph:

\begin{solution}

\eq{
	t_\ell &= \dfrac{480m}{0.8c} = 2\mu s \\
 	V_0^+ &= 20V\lp \dfrac{50\ohms}{125\ohms + 50\ohms}\rp = 5.7V \\
 	\Gamma_L &= \dfrac{75\ohms - 50\ohms}{125\ohms} = \dfrac{1}{5} \\
 	\Gamma_S &= \dfrac{125\ohms - 50\ohms}{175\ohms} = 0.429 \\
 	V_{\text{load}} (t=2\mu s) &= V_0(1+ \Gamma_L) =  5.7V(1 + 0.2) = 6.84V \\
 	V_{\text{load}} (t=6\mu s) &= V_0(1+\Gamma_L) (1+\Gamma_S\Gamma_L) = (5.7)(1.2)(1+(0.2)(0.429)) = 7.43V
}
