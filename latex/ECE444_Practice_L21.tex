\question \textbf{How the kit builds its sweep.} The ADALM-PHASER carries
$N = 8$ elements spaced $d = 14$ \text{mm} apart. At $10.3\ \ghz$ the
wavelength is $29.1$ \text{mm}, so $kd = 173.2\mydeg$ of phase per unit of
$\sin\theta$. In the Rectangular plot the kit holds the HB100 still and steps
the element-to-element phase one $2.8125\mydeg$ LSB per point.
\begin{parts}
	\begin{minipage}[t]{0.58\linewidth}
		\part Find the change in $\sin\theta_0$ for one step, and the step in
			angle near broadside.
		\begin{solution}[1.1in]
		\eq{ \Delta\lp\sin\theta_0\rp &= \frac{2.8125\mydeg}{173.2\mydeg} = 0.0162 \\
			\Delta\theta_0 &\approx 0.0162\ \text{rad} = 0.93\mydeg }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$\Delta\theta_0$ = \ansbox[1.3in]{$0.93\mydeg$}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the same step in angle near $30\mydeg$.
		\begin{solution}[1.1in]
		A fixed step in $\sin\theta$ is a larger step in $\theta$ where
		$\cosp{\theta}$ is small:
		\eq{ \Delta\theta_0 \approx \frac{0.93\mydeg}{\cosp{30\mydeg}} = 1.07\mydeg }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$\Delta\theta_0$ = \ansbox[1.3in]{$1.07\mydeg$}
	\end{minipage}

	\part Every point of the sweep comes from a different beam. In two or
		three sentences, explain why the trace still looks like the pattern of
		a single beam.
		\begin{solution}[1.4in]
		The array factor depends only on
		$\psi = kd\lp\sin\theta_\text{src} - \sin\theta_0\rp$, and its magnitude
		is the same for $\psi$ and $-\psi$. Stepping $\theta_0$ past a fixed
		source runs $\psi$ through the same values as moving the source past a
		beam held at a fixed $\theta_0$, so both draw the same curve. The
		sweep's x-axis is the commanded angle $\theta_0$, not a measured
		arrival angle.
		\end{solution}

	\part State one thing the sweep cannot show that a measurement with one
		beam held still can, and give the reason.
		\begin{solution}[1.2in]
		The pattern of each patch element. The sweep holds the source in one
		direction, so each patch receives the same level at every point, and
		that constant drops out of a trace read in dBc. A held beam with a
		moving source carries the source across each patch's pattern, which
		falls toward $\pm 90\mydeg$.
		\end{solution}
\end{parts}

\question \textbf{The walk.} With Lab Preset 7 the kit holds the beam at
$0\mydeg$ and plots the peak level against time while you carry the HB100
from $-90\mydeg$ to $+90\mydeg$ on a $1.2\ \m$ arc in $25$ s. The beam's
half-power width is $13.3\mydeg$.
\begin{parts}
	\begin{minipage}[t]{0.58\linewidth}
		\part At a constant pace, how long does the source spend inside the
			half-power width of the main lobe?
		\begin{solution}[1.1in]
		\eq{ \frac{180\mydeg}{25\ \text{s}} &= 7.2\mydeg\text{/s} \\
			t &= \frac{13.3\mydeg}{7.2\mydeg\text{/s}} = 1.85\ \text{s} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$t$ = \ansbox[1.3in]{$1.85$ s}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part You walk through the main lobe at $70\%$ of your average pace,
			then convert time to angle at the average pace. What half-power
			width does the trace report?
		\begin{solution}[1.1in]
		The source takes $1/0.7$ times as long to cross the lobe, so the lobe
		spans $1/0.7$ times as much time:
		\eq{ \theta_\text{HP} = \frac{13.3\mydeg}{0.7} = 19.0\mydeg }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$\theta_\text{HP}$ = \ansbox[1.3in]{$19.0\mydeg$}
	\end{minipage}

	\part Does the slow pace in part (b) change the level of the main lobe or
		of the first sidelobes on the trace? Explain in one or two sentences.
		\begin{solution}[1.1in]
		No. The level at each instant is the pattern in the direction the
		source occupies at that instant, whatever its speed, so a slow pace
		stretches a lobe in time without changing its height.
		\end{solution}

	\begin{minipage}[t]{0.58\linewidth}
		\part Your arm drifts $10$ \text{cm} outward, to $1.3\ \m$, as you walk.
			By how much does the received level change?
		\begin{solution}[1.1in]
		Received power falls as $1/r^2$:
		\eq{ 20\mylog{\frac{1.3\ \m}{1.2\ \m}} = 0.7\db }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$\Delta L$ = \ansbox[1.3in]{$-0.7\db$}
	\end{minipage}
\end{parts}

\question \textbf{Comparing the two traces.} A team records a sweep with the
HB100 at boresight and a walk with the beam held at $0\mydeg$, both at
$1.2\ \m$. The sweep reads its first sidelobes at $-12.6$ and $-13.4$ dBc.
The walk reads them at $-11.6$ and $-14.7$ dBc. Each reading carries about
$\pm 1\db$ from room reflections.
\begin{parts}
	\part Do the two methods agree on the first sidelobes? Give the reason,
		using the uncertainty of a dBc value.
		\begin{solution}[1.3in]
		A dBc value is the difference of two readings, each good to about
		$\pm 1\db$, so it is good to about $\pm 2\db$. Every one of the four
		values lies within $2\db$ of the array factor's $-12.8\db$, so the two
		methods agree. The $3.1\db$ spread between the walk's two sidelobes is
		about what reflections in a classroom produce.
		\end{solution}

	\part The team reads the walk's main-lobe width as $17\mydeg$ and the
		sweep's as $13\mydeg$. Which width do you report, and why?
		\begin{solution}[1.2in]
		The sweep's. Its x-axis is the commanded angle, good to about one
		$0.93\mydeg$ step, while the walk's x-axis is time, which becomes angle
		only at a constant pace. A slow pace through the main lobe widens it on
		the walk's trace, as in Question 2(b).
		\end{solution}

	\part Beyond $\pm 50\mydeg$ the sweep's lobes read about $-17$ dBc and the
		walk's about $-21$ dBc. Name two reasons the far lobes differ.
		\begin{solution}[1.3in]
		The sweep has a floor about $23\db$ below its peak, which adds to
		sidelobes that sit only a few decibels above it. The walk carries the
		HB100 across each patch's pattern, which falls toward $\pm 90\mydeg$
		because the ground plane blocks the back; the sweep holds the HB100 in
		one direction, so that pattern does not shape its trace. Reflections
		also matter most where the pattern is weak.
		\end{solution}

	\part The walk shows a lobe at $-20$ dBc near $+75\mydeg$ that is not in
		the sweep, and it moves when a classmate steps aside. What is it?
		\begin{solution}[1.0in]
		A reflection off the classmate. A feature of the antenna stays put when
		people move; a feature that moves with them comes from the room.
		\end{solution}
\end{parts}

\question \textbf{The beam at $\mathbf{30\mydeg}$.} The team now holds the beam
at $30\mydeg$ for the walk, and places the HB100 at $30\mydeg$ for the sweep.
For this array $\lambda/(Nd) = 0.260$ at $10.3\ \ghz$.
\begin{parts}
	\begin{minipage}[t]{0.58\linewidth}
		\part Find the angles of the two first nulls of the $30\mydeg$ beam.
		\begin{solution}[1.2in]
		The pattern slides in $\sin\theta$ (Lesson 18):
		\eq{ \sin\theta_\text{null} &= 0.5 \pm 0.260 = 0.240,\ 0.760 \\
			\theta_\text{null} &= 13.9\mydeg,\ 49.5\mydeg }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$\theta_\text{null}$ = \ansbox[1.3in]{$13.9\mydeg,\ 49.5\mydeg$}
	\end{minipage}

	\part Explain why the main lobe reaches farther above its peak than below
		it, and whether the sweep and the walk should both show this.
		\begin{solution}[1.3in]
		The nulls are an equal step of $0.260$ apart in $\sin\theta$, and a step
		in $\sin\theta$ spans more angle where $\cosp{\theta}$ is small, so the
		lobe reaches $19.5\mydeg$ above its peak and $16.1\mydeg$ below. Both
		methods run $\psi$ through the same values, so both show it.
		\end{solution}

	\part The array factor puts both first sidelobes of this beam at
		$-12.8$ dBc. The walk reads the outer one, near $60\mydeg$, at about
		$-15$ dBc, while the sweep reads it near $-12$ dBc. Give the cause, and
		say whether one walk in a classroom can confirm it.
		\begin{solution}[1.5in]
		The walk carries the HB100 out to $60\mydeg$, where each patch receives
		about $3\db$ less than at broadside; the sweep keeps the HB100 at
		$30\mydeg$ for every point, so the patch pattern does not change its
		shape. One walk cannot confirm it: the room alone can make two first
		sidelobes differ by about $3\db$, as large as the effect. Repeated
		turns of the kit on its stand, at a fixed distance, are the better test.
		\end{solution}
\end{parts}

\question \textbf{Source distance.} Lesson 18 takes the course array's aperture as
$D = Nd = 112$ \text{mm}. You place the HB100 $1.2\ \m$ from the array face.
\begin{parts}
	\begin{minipage}[t]{0.58\linewidth}
		\part Compute the far-field distance for this array at $10.3\ \ghz$
			($\lambda = 29.1$ \text{mm}), and state whether the lab range
			qualifies.
		\begin{solution}[1.2in]
		\eq{ r \ge \frac{2D^2}{\lambda}
			= \frac{2\lp0.112\ \m\rp^2}{0.0291\ \m} = 0.86\ \m }
		The $1.2\ \m$ range is past this, so the edge-to-center phase error of
		the incoming wavefront stays under the $22.5\mydeg$ the criterion allows.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$r$ = \ansbox[1.3in]{$0.86\ \m$}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part You need to measure a larger array on the same $1.2\ \m$ range at
			$10.3\ \ghz$. Find the largest aperture dimension that still satisfies
			the far-field criterion.
		\begin{solution}[1.2in]
		Solve the criterion for $D$:
		\eq{ D = \sqrt{\frac{r\lambda}{2}}
			= \sqrt{\frac{\lp1.2\ \m\rp\lp0.0291\ \m\rp}{2}} = 0.132\ \m }
		so an aperture larger than about $132$ \text{mm} needs a longer range.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$D$ = \ansbox[1.3in]{$132$ \text{mm}}
	\end{minipage}

	\part Name two changes to the setup, not the antenna, that make the far
		sidelobes of either trace more trustworthy.
		\begin{solution}[1.2in]
		Reduce the reflections: put absorber on the bench and nearby surfaces,
		keep people and equipment out of the path, and set the array and the
		HB100 at the same height away from large flat surfaces. Then repeat the
		measurement with the whole setup moved; a feature that moves with the
		room comes from a reflection, and one that stays with the array comes
		from the antenna.
		\end{solution}
\end{parts}
